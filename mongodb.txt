to install lila
git clone --recursive https://github.com/lichess-org/lila.git
cd lila
corepack enable
mongosh lichess < bin/mongodb/indexes.js
ui/build
./lila.sh

edit lila/modules/web/src/main/helper/AssetFullHelper.scala
if netConfig.siteName == "liladocker.duckdns.org:9663" then "lichess.dev"

edit lila/conf/base.conf
domain = "liladocker.duckdns.org:9663"
socket.domains = ["liladocker.duckdns.org:9664"]
ratelimit = false

git clone https://github.com/lichess-org/lila-ws.git
cd lila-ws
sbt run -Dcsrf.origin=http://liladocker.duckdns.org:9663

to install homebrew
NONINTERACTIVE=1 /bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/uninstall.sh)"
sudo rm -rf /opt/homebrew
/bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"
